```python
def breadth_series(m: pd.DataFrame) -> pd.Series
    """up/down（down==0 → inf 档标记）；m 按 trade_date 升序。"""

def breadth_distribution(br: pd.Series) -> Dict[str, Any]
    """分位表 P50/P75/P90/P95/P99 + ≥{1.5,2.0,2.5} 历史占比 + inf 档天数。"""

def conditional_next_day(m: pd.DataFrame, sh: pd.DataFrame) -> List[Dict]
    """分桶 [0,1)/[1,1.5)/[1.5,2)/[2,2.5)/[2.5,4)/[4,inf)+inf × 次日结果：
    次日溢价均值/涨停家数中位变化/上证次日开盘 gap 与收盘涨幅中位；
    m 内 shift(-1) 对齐，sh 按 trade_date merge（缺日剔除样本并计数）。"""

def incremental_power(m: pd.DataFrame, climax_core: bool...) -> Dict
    """条件增量（第二证据存在意义）：core=max_streak≥5 且 premium_mean≥4——
    (a) core 命中日内 breadth≥2 vs <2 的次日差异；
    (b) core 未命中日中 breadth 越各阈值档的次日表现（极端宽度单独预告力）。"""

def threshold_options(m: pd.DataFrame, thresholds=(1.5, 2.0, 2.5, 'P90')) -> List[Dict]
    """各阈值：历史命中天数/占比、次日溢价、次日上证涨幅中位、与 core 交叠度。"""
```

```python
def align_overseas(sh_days: pd.Series, ov: pd.DataFrame) -> pd.DataFrame
    """逐 A 股交易日 D：ov 收盘 as-of 前向填充（last ≤ D 与 last ≤ D-1）；
    D 无前日（首日）剔除。A50 休市日自动前向填充（如周一用得上周五夜盘 bar）。
    注意 D 日的 bar 含 D 日盘后信息（'全天收'代理——报告口径句必须标注乐观界）。"""

def consistency(merged: pd.DataFrame) -> Dict
    """同向率 = P(sign(ov_move) == sign(上证 open_gap))：全样本/分年度/
    周一(周末缝)子样本；open_gap = open/preclose − 1（上证 D 日）。"""

def trigger_availability(merged_mon: pd.DataFrame, thresholds=(1.0,1.5,2.0)) -> Dict
    """仅周五锚定→周一子样本：|A50 Fri→Mon 变动| 越各阈频度；越阈后
    P1 命中率（≤−thr → 上证周一 open<0）/ P3 命中率（≥+thr → open>0）；
    同表并列 hkHSI/usIXIC 同口径对照（A50 相对增量的来源）。"""
```

**A5 准入门指标计算口径（与 V4.0 §2.X3 三门一一对应）**：

## 11. 风险与回滚

- **风险 1**（A2）：`_r1_raw_hit` 与链侧 `r1_ladder_relay` 谓词输入不一致
  （如链侧 open_count 来源不同）→ 对账验收（20261023 三列逐一对账）兜底；
  不一致即修输入对齐。
- **风险 2**（A2）：hermetic 直测 IO 触点比预期多（_score_dims/_fundamental 等）
  → 实例级 monkeypatch 逐个隔离；仍不可隔离则退回 phase3 式 stub reader 合成
  agg 路径（记录取舍）。
- **风险 3**（A1/A2）：缓存版本号编辑破坏类属性缩进 → 提交前 `py_compile`
  三件套（v9 时踩过）。
- **风险 4**（A4）：A50 日K 断档/迟更新 → align 前向填充 + 样本不足（<250 根）
  降级不产结论（§5.4）。
- **回滚**：每 A 步独立 commit，单步 `git revert`；A3/A4 纯增量（新文件+只读），
  删文件即回滚；无 DB 迁移、无配置变更。
