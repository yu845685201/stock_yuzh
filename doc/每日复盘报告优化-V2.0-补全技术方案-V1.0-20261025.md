```python
def hit_concept_top(top_concepts: List[str], concepts: List[str]) -> Optional[str]:
    """命中概念取聚集序最前（聚集序 = 概念内当日涨停家数降序）。

    P3-4（C9）：is_r2 已保证交集非空，default=None 仅为防御——若未来重构
    移除该前置保护，此处退化为不命中（不硬造）而非抛 StopIteration。
    """
    return next((c for c in top_concepts if c in set(concepts)), None)
```

```python
def _anchor_conf_note(ca: Optional[Dict[str, Any]]) -> Optional[str]:
    """P1-6 资金锚置信度一行（技术方案 §4.6 落点对齐）：标签与综合置信度
    推导单一出处 = aggregator._capital_anchors（anchor_tags/confidence），

    data = (ca or {}).get('data') or {}
    tags, conf = data.get('anchor_tags') or {}, data.get('confidence')
    if not tags or not conf:
        return None
    items = ' · '.join(f'{lbl}={tags.get(key, "—")}'
                       for lbl, key in (('北向成交额', 'north_total'),
                                        ('南向净买入', 'south_net'),
                                        ('ETF份额', 'etf')))
    return (f'资金锚置信度：综合 {conf}（{items}；停旧值=快照日期早于锚定日'
            '按停更对待；明细见六维报告 §一）')
```
