**(b) 提取函数**（纯函数，免库可测）：

```python
_RE_TRUNK = re.compile(r'\*\*主干：(.+?)\*\*\s*$')
_RE_BIAS = re.compile(r'\*\*基准倾向\*\*：\s*(.+)$')

def _extract_tldr_fields(note_text: Optional[str]) -> Dict[str, Optional[str]]:
    """速览三要素之 note 侧两要素：主干句与基准倾向行（契约见 SKILL 综合研判①
    与预案节；提取失败返回 None——由 check ⑥⑦ 在发布侧响亮阻断，此处静默降级
    是预期行为：无 note 日速览框退化为仅关键数行）。"""
    trunk = bias = None
    for line in (note_text or '').splitlines():
        s = line.strip().lstrip('- ')
        if trunk is None:
            m = _RE_TRUNK.match(s)
            if m:
                trunk = m.group(1).strip()
        if bias is None:
            m = _RE_BIAS.match(s)
            if m:
                bias = m.group(1).strip()
    return {'trunk': trunk, 'bias': bias}
```
